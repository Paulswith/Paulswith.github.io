def change\_to\_duration(position\_list):

    """

    勾股定理 求d=C=开根号(A^2+B^2)

    """

    if not position\_list:

        return

    x1,y1 = position\_list\[0\]

    x2,y2 = position\_list\[1\]

    the\_d = math.sqrt(math.pow(y2 - y1, 2) + math.pow(x2 - x1, 2))

    return float('%.3f' % (the\_d \* each\_pixel\_d))
